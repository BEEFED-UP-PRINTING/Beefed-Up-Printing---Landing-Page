import { useEffect, useRef, useState, useCallback } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { ChevronDown } from "lucide-react";

interface HeroProps {
  onLightsOn: () => void;
}

type Phase = "loading" | "lit";

export default function Hero({ onLightsOn }: HeroProps) {
  const [phase, setPhase] = useState<Phase>("loading");
  const lightsOnFired = useRef(false);

  const artworkSrc = `${import.meta.env.BASE_URL}skyline-placeholder.png`;
  const litArtworkSrc = `${import.meta.env.BASE_URL}file_00000000cd1871fbb989bbb0aecb1b7f.png`;

  const fireLightsOn = useCallback(() => {
    if (lightsOnFired.current) return;
    lightsOnFired.current = true;
    onLightsOn();
  }, [onLightsOn]);

  useEffect(() => {
    const timer = setTimeout(() => {
      if (phase === "loading") {
        setPhase("lit");
        fireLightsOn();
      }
    }, 6000); // shorter, better UX

    return () => clearTimeout(timer);
  }, [phase, fireLightsOn]);

  const imgContain =
    "absolute inset-0 w-full h-full object-contain px-2 sm:px-4 md:px-12 pt-10 sm:pt-8 md:pt-0";

  return (
    <section className="relative w-full h-[100svh] bg-black flex flex-col items-center justify-center overflow-hidden">

      {/* Fade overlay */}
      <motion.div
        initial={{ opacity: 1 }}
        animate={{ opacity: phase === "loading" ? 1 : 0 }}
        transition={{ duration: 1.2 }}
        className="absolute inset-0 bg-black z-10"
      />

      {/* Dark artwork */}
      <motion.img
        src={artworkSrc}
        alt=""
        aria-hidden
        initial={{ opacity: 0 }}
        animate={{ opacity: phase === "loading" ? 1 : 0 }}
        transition={{ duration: 0.35 }}
        className={imgContain}
        style={{ zIndex: 21, objectPosition: "center top" }}
      />

      {/* Lit artwork (ONLY boombox now) */}
      <motion.img
        src={litArtworkSrc}
        alt="Beefed Up Printing — Street Culture Custom Merch"
        initial={{ opacity: 0 }}
        animate={{ opacity: phase === "lit" ? 1 : 0 }}
        transition={{ duration: 0.7 }}
        className={imgContain}
        style={{ zIndex: 12, objectPosition: "center top" }}
      />

      {/* Gradients */}
      <div className="absolute inset-0 pointer-events-none z-20 bg-[radial-gradient(ellipse_at_center,transparent_38%,rgba(0,0,0,0.92)_100%)]" />
      <div className="absolute inset-0 pointer-events-none z-20 bg-[linear-gradient(to_bottom,rgba(0,0,0,0.65)_0%,transparent_22%,transparent_75%,rgba(0,0,0,0.85)_100%)]" />

      {/* Scroll indicator */}
      <AnimatePresence>
        {phase === "lit" && (
          <motion.div
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            transition={{ delay: 1.2 }}
            className="absolute bottom-8 left-1/2 -translate-x-1/2 flex flex-col items-center gap-2 z-30"
          >
            <span className="text-[10px] uppercase tracking-[0.3em] text-zinc-500 font-bold">
              Scroll
            </span>
            <a href="#about" className="text-zinc-500 hover:text-orange-400 transition animate-bounce">
              <ChevronDown size={28} />
            </a>
          </motion.div>
        )}
      </AnimatePresence>
    </section>
  );
}
🔥 2. AudioPlayer.tsx (NO UI — invisible player)

import { useEffect, useImperativeHandle, useRef, forwardRef, useCallback } from "react";

export interface AudioPlayerHandle {
  triggerPlay: () => void;
}

const AudioPlayer = forwardRef<AudioPlayerHandle>((_, ref) => {
  const audioRef = useRef<HTMLAudioElement>(null);
  const unlockedRef = useRef(false);
  const pendingRef = useRef(false);

  const unlock = useCallback(() => {
    if (unlockedRef.current) return;
    unlockedRef.current = true;

    const audio = audioRef.current;
    if (!audio) return;

    audio.muted = true;

    audio.play()
      .then(() => {
        if (pendingRef.current) {
          audio.muted = false;
        }
      })
      .catch(() => {});
  }, []);

  useEffect(() => {
    unlock();

    const events = ["click", "touchstart", "keydown", "scroll"] as const;
    const handler = () => unlock();

    events.forEach(e =>
      window.addEventListener(e, handler, { once: true, passive: true })
    );

    return () =>
      events.forEach(e => window.removeEventListener(e, handler));
  }, [unlock]);

  useImperativeHandle(ref, () => ({
    triggerPlay() {
      pendingRef.current = true;

      const audio = audioRef.current;
      if (!audio) return;

      if (unlockedRef.current) {
        audio.muted = false;
        audio.play().catch(() => {});
      }
    },
  }));

  return (
    <audio
      ref={audioRef}
      src={`${import.meta.env.BASE_URL}music.mp3`}
      loop
      preload="auto"
    />
  );
});

AudioPlayer.displayName = "AudioPlayer";
export default AudioPlayer;

import { useRef, useCallback } from "react";
import Navbar from "@/components/Navbar";
import Hero from "@/components/Hero";
import GreetingsTicker from "@/components/GreetingsTicker";
import About from "@/components/About";
import Services from "@/components/Services";
import ContactForm from "@/components/ContactForm";
import Footer from "@/components/Footer";
import AudioPlayer, { type AudioPlayerHandle } from "@/components/AudioPlayer";

export default function Home() {
  const audioPlayerRef = useRef<AudioPlayerHandle>(null);

  const handleLightsOn = useCallback(() => {
    setTimeout(() => {
      audioPlayerRef.current?.triggerPlay();
    }, 400); // slight cinematic delay
  }, []);

  return (
    <div className="bg-background min-h-screen text-foreground relative">
      <Navbar />

      <main>
        <Hero onLightsOn={handleLightsOn} />
        <GreetingsTicker />
        <About />
        <Services />
        <ContactForm />
      </main>

      <Footer />

      {/* Invisible audio controller */}
      <AudioPlayer ref={audioPlayerRef} />
    </div>
  );
}
